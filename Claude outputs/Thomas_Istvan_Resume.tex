% Section headers: bold caps, with a thin rule underneath (matches the
% reference resume's convention). Plain \bfseries rather than \bfseries +
% \scshape — Computer Modern has no true bold-small-caps shape, so that
% combination silently substitutes fonts. Using titlesec's own \titlerule
% (rather than a manual \hrule + negative \vspace) so the rule always sits
% cleanly below the text instead of risking an overlap/strikethrough look.
\titleformat{\section}
  {\normalsize\bfseries}
  {}{0em}
  {}
  [{\color{rulegray}\titlerule[0.6pt]}]

% everywhere.

}
